cmake_minimum_required(VERSION 3.24)
project(ShadowPC_Engine_v14 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SHADOWPC_ENABLE_DX12 "Build Direct3D 12 backend" ${WIN32})
option(SHADOWPC_ENABLE_VULKAN "Build Vulkan backend" OFF)
add_library(shadowpc_core
 src/core/Allocator.cpp src/core/AssetRegistry.cpp src/core/DependencyTracker.cpp src/core/RegistryBinary.cpp src/core/ResourceManager.cpp src/core/VirtualArena.cpp
 src/assets/AssetDecoder.cpp src/assets/AssetPipeline.cpp
 src/streaming/AsyncIo.cpp src/streaming/Decompression.cpp src/streaming/WorldStreaming.cpp src/streaming/PriorityEngine.cpp
 src/rhi/NullBackend.cpp src/rhi/UploadScheduler.cpp src/rhi/GpuResidency.cpp)
target_include_directories(shadowpc_core PUBLIC include)
if(MSVC)
  target_compile_options(shadowpc_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(shadowpc_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(SHADOWPC_ENABLE_DX12)
  if(NOT WIN32)
    message(FATAL_ERROR "DX12 backend requires Windows")
  endif()
 add_library(shadowpc_dx12 src/rhi/Dx12Backend.cpp)
target_include_directories(shadowpc_dx12 PUBLIC include)
target_link_libraries(shadowpc_dx12 PUBLIC shadowpc_core d3d12 dxgi dxguid)
endif()
if(SHADOWPC_ENABLE_VULKAN)
 find_package(Vulkan REQUIRED)
 add_library(shadowpc_vulkan src/rhi/VulkanBackend.cpp)
target_include_directories(shadowpc_vulkan PUBLIC include)
target_link_libraries(shadowpc_vulkan PUBLIC shadowpc_core Vulkan::Vulkan)
endif()
add_executable(shadowpc_core_tests tests/test_core.cpp)
target_link_libraries(shadowpc_core_tests PRIVATE shadowpc_core)
enable_testing()
add_test(NAME core COMMAND shadowpc_core_tests)
